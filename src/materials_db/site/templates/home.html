import lightsmith as ls

# any dataset in this database
au = ls.fetch_refractiveindex("main", "Au", "Johnson")

sim = ls.Simulation(
    particles=[ls.Particle(shape=ls.Sphere(radius_nm=40), material=au)],
    medium_index=1.33,
    wavelengths=ls.WavelengthRange(start_nm=400, stop_nm=800, num=81),
)
res = ls.solve(sim)   # exact Mie, or DDA, FDTD, BEM…
